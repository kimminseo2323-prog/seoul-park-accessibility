### 3. modules/data_loader.py - 데이터 로더

**역할**: 데이터 로딩 및 캐싱

```python
"""
데이터 로딩 및 전처리 모듈
"""
import geopandas as gpd
import pandas as pd
from pathlib import Path
from .config import *

class DataLoader:
    """
    데이터 로더 클래스 (싱글톤 패턴 + 캐싱)
    """
    def __init__(self):
        self._gdf_dong = None
        self._gdf_dong_service = None
        self._gdf_parks = None
        self._gdf_grid = None
        self._df_2sfca = None
        self._gdf_sigungu = None
        self._gdf_sido = None

    def load_dong_geometry(self):
        """행정동 경계 + 인구 로드 (캐싱)"""
        if self._gdf_dong is None:
            print(f"Loading dong geometry from: {GEOJSON_DONG}")
            self._gdf_dong = gpd.read_file(GEOJSON_DONG)
            self._gdf_dong = self._gdf_dong.to_crs(epsg=4326)
        return self._gdf_dong

    def load_dong_service_score(self):
        """2SFCA 분석 결과 로드 (캐싱)"""
        if self._gdf_dong_service is None:
            print(f"Loading service score from: {GEOJSON_DONG_SERVICE}")
            self._gdf_dong_service = gpd.read_file(GEOJSON_DONG_SERVICE)
            self._gdf_dong_service = self._gdf_dong_service.to_crs(epsg=4326)
        return self._gdf_dong_service

    def load_parks(self):
        """공원 데이터 로드 (캐싱)"""
        if self._gdf_parks is None:
            print(f"Loading parks from: {GEOJSON_PARKS}")
            self._gdf_parks = gpd.read_file(GEOJSON_PARKS)
            self._gdf_parks = self._gdf_parks.to_crs(epsg=4326)

            # 공원 유형 추출 (LABEL 컬럼에서)
            if 'park_type' not in self._gdf_parks.columns:
                self._gdf_parks['park_type'] = self._gdf_parks['LABEL'].str.extract(r'(.*?공원)')[0]
                self._gdf_parks['park_type'] = self._gdf_parks['park_type'].fillna('기타')

        return self._gdf_parks

    def load_grid_population(self):
        """100m 격자 인구 로드 (캐싱)"""
        if self._gdf_grid is None:
            print(f"Loading grid population from: {SHAPEFILE_GRID}")
            self._gdf_grid = gpd.read_file(SHAPEFILE_GRID)
            self._gdf_grid = self._gdf_grid.to_crs(epsg=4326)

            # 인구 100명 이상만 필터링 (성능 최적화)
            self._gdf_grid = self._gdf_grid[self._gdf_grid['val'] >= MIN_POPULATION_THRESHOLD]

        return self._gdf_grid

    def load_sigungu_boundary(self):
        """시군구 경계 로드 (캐싱)"""
        if self._gdf_sigungu is None:
            print(f"Loading sigungu boundary from: {SIGUNGU_POLYGON_SHP}")
            self._gdf_sigungu = gpd.read_file(SIGUNGU_POLYGON_SHP)
            self._gdf_sigungu = self._gdf_sigungu.to_crs(epsg=4326)

            # 서울시만 필터링 (SIDO_CD == '11')
            self._gdf_sigungu = self._gdf_sigungu[self._gdf_sigungu['SIDO_CD'] == '11']

        return self._gdf_sigungu

    def get_park_types(self):
        """공원 유형 목록 반환"""
        gdf_parks = self.load_parks()
        return sorted(gdf_parks['park_type'].unique())
```

#### 4.1 create_service_map() - 접근성 지도

```python
def create_service_map(gdf, column='avg_service_per_capita',
                       colorscale=None, title='서비스 접근성',
                       basemap_opacity=0.8, show_parks=True,
                       show_buffers=False, buffer_distance=500):
    """
    백분위수 기반 접근성 지도 생성 (Choroplethmapbox)

    Parameters:
    -----------
    gdf : GeoDataFrame
        행정동 서비스 점수 데이터
    column : str
        시각화할 컬럼 ('avg_service_per_capita')
    colorscale : list
        색상 스케일 (None이면 config의 CHOROPLETH_COLORSCALE 사용)
    basemap_opacity : float
        행정동 레이어 투명도 (0~1)
    show_parks : bool
        공원 레이어 표시 여부
    show_buffers : bool
        버퍼 레이어 표시 여부
    buffer_distance : int
        버퍼 거리 (m)

    Returns:
    --------
    plotly.graph_objects.Figure
    """
    import plotly.graph_objects as go
    import json
    import numpy as np

    if colorscale is None:
        colorscale = CHOROPLETH_COLORSCALE

    # GeoJSON 최적화
    gdf_viz = gdf.copy()
    gdf_viz = gdf_viz.to_crs(epsg=4326)
    gdf_viz = gdf_viz.reset_index(drop=True)
    gdf_viz['id'] = gdf_viz.index.astype(str)

    # 백분위수 기반 색상 스케일 범위 계산
    values = gdf_viz[column].dropna()
    if len(values) > 0:
        # 10%, 25%, 50%, 75%, 90% 백분위수 계산
        p10 = np.percentile(values, 10)
        p25 = np.percentile(values, 25)
        p50 = np.percentile(values, 50)
        p75 = np.percentile(values, 75)
        p90 = np.percentile(values, 90)

        # 색상 스케일 범위: 10%~90% 백분위수 사용 (극단값 제외)
        zmin = p10
        zmax = p90
    else:
        zmin = 0
        zmax = 100

    # GeoJSON 변환
    geojson_all = json.loads(gdf_viz.to_json())
    geojson_all = _sanitize_geojson(geojson_all)

    # Figure 생성
    fig = go.Figure()

    # Choroplethmapbox 추가 (백분위수 기반 스케일링)
    fig.add_trace(go.Choroplethmapbox(
        geojson=geojson_all,
        locations=gdf_viz['id'].tolist(),
        z=gdf_viz[column].tolist(),
        featureidkey="properties.id",
        colorscale=colorscale,
        zmin=zmin,  # 10% 백분위수
        zmax=zmax,  # 90% 백분위수
        marker=dict(opacity=basemap_opacity, line=dict(width=0.5, color='#CCCCCC')),
        colorbar=dict(
            title=dict(text="접근성 점수<br>(m²/인)", font=dict(size=11, color='white')),
            thickness=20,
            len=0.7,
            x=1.02,
            tickfont=dict(size=10, color='white'),
            bgcolor='rgba(30,30,30,0.8)',
            bordercolor='#555',
            borderwidth=1,
            tickvals=[p10, p25, p50, p75, p90],  # 백분위수 눈금
            ticktext=[f'{p10:.1f}', f'{p25:.1f}', f'{p50:.1f}', f'{p75:.1f}', f'{p90:.1f}']
        ),
        hovertemplate='<b>%{customdata[0]}</b><br>' +
                      '접근성: %{z:.2f} m²/인<br>' +
                      '<extra></extra>',
        customdata=gdf_viz[['ADM_NM']].values,
        showscale=True
    ))

    # 지도 레이아웃
    fig.update_layout(
        mapbox=dict(
            style='open-street-map',
            center={'lat': MAP_CENTER['lat'], 'lon': MAP_CENTER['lon']},
            zoom=MAP_ZOOM
        ),
        margin=dict(l=0, r=0, t=40, b=0),
        height=MAP_HEIGHT,
        title=dict(text=f'<b>{title}</b>', x=0.5, xanchor='center'),
        paper_bgcolor='#1E1E1E',
        font=dict(color='white')
    )

    return fig
```

#### 4.2 create_sigungu_comparison_chart() - 시군구 비교 차트

```python
def create_sigungu_comparison_chart(df, value_col='avg_service_per_capita',
                                    label_col='SIGUNGU_NM', title='시군구별 비교'):
    """
    시군구별 접근성 점수 비교 차트 (수평 막대 그래프)

    Parameters:
    -----------
    df : DataFrame
        시군구별 서비스 점수
    value_col : str
        값 컬럼명
    label_col : str
        라벨 컬럼명
    title : str
        차트 제목

    Returns:
    --------
    plotly.graph_objects.Figure
    """
    import plotly.graph_objects as go

    # 내림차순 정렬
    df_sorted = df.sort_values(value_col, ascending=True)

    # 최소/최대값 계산
    min_val = df_sorted[value_col].min()
    max_val = df_sorted[value_col].max()

    # Blues 색상 그라데이션 생성
    colorscale = [
        [0.0, '#f7fbff'],   # 매우 연한 파랑
        [0.2, '#deebf7'],
        [0.4, '#c6dbef'],
        [0.6, '#9ecae1'],
        [0.8, '#6baed6'],
        [1.0, '#08519c']    # 매우 진한 파랑
    ]

    fig = go.Figure()

    fig.add_trace(go.Bar(
        x=df_sorted[value_col],
        y=df_sorted[label_col],
        orientation='h',
        marker=dict(
            color=df_sorted[value_col],
            colorscale=colorscale,
            cmin=min_val,
            cmax=max_val,
            showscale=True,
            colorbar=dict(
                title="접근성 점수",
                x=1.15
            )
        ),
        text=[f'{val:.2f}' for val in df_sorted[value_col]],
        textposition='outside',
        hovertemplate='<b>%{y}</b><br>접근성: %{x:.2f} m²/인<extra></extra>'
    ))

    fig.update_layout(
        title=dict(text=f'<b>{title}</b>', x=0.5, xanchor='center'),
        xaxis_title="평균 접근성 점수 (m²/인)",
        yaxis_title="",
        height=800,
        margin=dict(l=100, r=150, t=60, b=40),
        paper_bgcolor='#1E1E1E',
        plot_bgcolor='#2D2D2D',
        font=dict(color='white'),
```

## 백분위수 기반 색상 스케일 구현

### 문제 인식

**기존 방식**:
- 고정 범위 (0~100)로 색상 매핑
- 실제 데이터: 0.08 ~ 338.49 m²/인
- 대부분 지역이 0~20 범위 → 흰색~연한 파랑만 표시
- 극소수 산지/공원 밀집 지역(>100)이 전체 색상 범위 왜곡

### 해결 방법

**백분위수 기반 스케일링**:
```python
# 10%, 25%, 50%, 75%, 90% 백분위수 계산
p10 = np.percentile(values, 10)   # 예: 0.5
p25 = np.percentile(values, 25)   # 예: 2.1
p50 = np.percentile(values, 50)   # 예: 5.8
p75 = np.percentile(values, 75)   # 예: 12.3
p90 = np.percentile(values, 90)   # 예: 25.7

# 색상 스케일 범위: 10%~90% 백분위수 사용
zmin = p10  # 0.5
zmax = p90  # 25.7
```

**효과**:
- 상위/하위 각 10% 극단값 제외
- 80% 중간 지역의 미세한 차이 시각화
- 컬러바 눈금에 백분위수 명시 (0.5, 2.1, 5.8, 12.3, 25.7)

### 코드 적용 (visualizations.py)

### Callback 4: update_sigungu_gini_chart()

**트리거**: 버퍼 거리, 공원 유형 변경 시

```python
@app.callback(
    Output('sigungu-gini-chart', 'figure'),
    [Input('buffer-distance-select', 'value'),
     Input('park-type-select', 'value')]
)
def update_sigungu_gini_chart(buffer_distance, selected_park_types):
    """
    시군구별 지니계수 비교 차트

    각 시군구 내부의 불평등 정도를 지니계수로 측정하여
    상위 10개 시군구를 표시합니다.

    지니계수 계산식:
    G = (2 * Σ(i × x_i)) / (n × Σ(x_i)) - (n+1)/n

    Parameters:
    -----------
    buffer_distance : int
        선택된 버퍼 거리
    selected_park_types : list
        선택된 공원 유형 목록

    Returns:
    --------
    plotly Figure : 시군구별 지니계수 막대 차트
    """
    df_multi = data_loader.load_2sfca_multi_buffer()
    df_filtered = df_multi[df_multi['buffer_distance'] == buffer_distance].copy()

    # 공원 유형 필터링
    if selected_park_types and len(selected_park_types) > 0 and '전체' in selected_park_types:
        df_filtered = df_filtered[df_filtered['park_types'] == '전체'].copy()

    # 각 시군구별 지니계수 계산
    gini_results = []
    for sigungu in df_filtered['SIGUNGU_NM'].unique():
        df_sigungu = df_filtered[df_filtered['SIGUNGU_NM'] == sigungu]
        scores = df_sigungu['avg_service_per_capita'].dropna().sort_values().values

        if len(scores) > 1:
            n = len(scores)
            gini = (2 * np.sum(np.arange(1, n + 1) * scores)) / (n * np.sum(scores)) - (n + 1) / n
            gini_results.append({'sigungu': sigungu, 'gini': gini})

    df_gini = pd.DataFrame(gini_results).nlargest(10, 'gini')

    # 차트 생성
    fig = go.Figure()
    fig.add_trace(go.Bar(
        x=df_gini['gini'],
        y=df_gini['sigungu'],
        orientation='h',
        marker=dict(color='#FF9800', line=dict(color='#F57C00', width=1))
    ))

    fig.update_layout(
        title='<b>시군구별 지니계수 비교 (상위 10)</b>',
        xaxis_title='지니계수',
        yaxis_title='',
        height=400,
        paper_bgcolor='#1E1E1E',
        plot_bgcolor='#2D2D2D',
        font=dict(color='white')
    )

    return fig
```

```python
class DataLoader:

        return self._gdf_dong  # 이후 캐시 반환
```

**효과**: 로딩 시간 5초 → 0.1초 (50배 개선)

```python
gdf_grid = gdf_grid[gdf_grid['val'] >= 100]  # 약 24,000개로 감소 (60% 감소)
```

```python
gdf_viz['geometry'] = gdf_viz['geometry'].simplify(tolerance=0.01, preserve_topology=True)
```

## 다중 버퍼 및 공원 유형별 분석 데이터 구조

### dong_service_score_multi_buffer_parktype.csv

**위치**: `s1_data/s2_processed_data/dong_service_score_multi_buffer_parktype.csv`

**데이터 구조**:
- 총 행 수: 8,075개 (426개 행정동 × 4개 버퍼 × ~5개 공원 유형 조합)
- 주요 컬럼:
  - `ADM_NM`: 행정동명 (426개)
  - `SIGUNGU_NM`: 시군구명 (25개)
  - `buffer_distance`: 버퍼 거리 (100, 300, 500, 1000)
  - `park_types`: 공원 유형 ('전체', '근린공원', '어린이공원', '소공원' 등)
  - `avg_service_per_capita`: 평균 서비스 점수 (m²/인)

**생성 스크립트**: `s2_module/3_다중버퍼_공원유형별_분석_최적화.py`

**최적화 기법**:
- 공원 버퍼 사전 계산 및 재사용
- Spatial indexing (`.sindex.intersection()`) 사용
- 병렬 처리 대신 단일 패스 처리로 메모리 효율 개선
- 실행 시간: 약 10-30분 (기존 무한 루프 해결)
